# Settings for "Play Rush of Blood.bat", one key=value a line. Lines starting with # are ignored;
# what is commented out below shows the value that applies when the line is left out.

# The window at the start (menu=0 to go without it) sets the field of view and a few more below.

# The size each eye is drawn at, in pixels of width: 1152 is the console's own (the picture of
# a PlayStation 4, 1152x1296 an eye); the larger, the sharper in the headset, and the more
# work for the graphics card and the more memory the game takes (1536 nearly doubles the
# pixels, 2048 triples them, 2880 is six times and needs about 3.5 GB more memory). 2304 is
# the default (the Medium choice of the window), 2880 the largest (High). 1152 to 2880; the
# window sets it with a slider.
#eye_width=2304

# Not used by this game: it picks its picture by itself, and does not draw smaller where the
# graphics card falls behind. (The line stays so that older settings files remain valid.)
#dynamic=1

# Not used by this game yet: it keeps in step with the headset's refresh rate, which Virtual
# Desktop sets (Streaming > Frame rate).
#fps=60

# The frame rate the game is held to, 20 to 60 (0 = the game's own timing, 19 to 26 a second in
# busy scenes). The emulator changes how often the game is told the headset has refreshed until
# the game runs at this rate; the window sets it. 60 is the game's own rate and the default:
# it is even with Virtual Desktop's frame rate at 120 (Streaming tab); use 45 at 90 and 40 at 80.
# Anything above what the PC can do just means as fast as it can.
#target_fps=60

# How much of the headset's field of view the game draws, in percent, 70 to 160: at 100 the
# picture fills all that the headset shows; less puts the same pixels over fewer degrees, a
# sharper picture with a dark border where the rest was; more draws a wider view (softer, and
# the coloured corners outside the lens circle move out of sight). 120 is the default: the
# picture looks smoothest there, with no borders (the corners outside the lens circles are
# black, see mask below). Without the mask, a Quest 3 needs 155 to hide them.
#fov=120

# What fov is a percent of: headset (the headset being worn, whatever it is: the emulator asks
# it as the game starts) or psvr (a PlayStation VR's 100 by 103 degrees an eye, the view the
# game was made for, which leaves a narrow dark margin far out to the sides on a Quest 3).
#fov_of=psvr

# 1: the window with the main settings comes up at every start.
#menu=1

# Sharpening of the picture on its way to the headset, 0 (none) to 1. Virtual Desktop has a
# sharpening of its own (Streaming tab): what is here comes on top of it.
#sharpen=0.3

# Black instead of the coloured corners outside the two lens circles of the picture, so that
# fov can be set low (about the circle's size: 100 to 110) for a sharper picture without the
# corners showing. 1 = the circles as drawn, 0 = off. If the picture's edge gets cut off, give
# a bit more (1.05); if some colour is left at the edge, a bit less (0.97).
#mask=1

# Anisotropic filtering of the game's textures, 0 (as the game draws them), 2, 4, 8 or 16. It
# keeps walls and floors seen at a slant sharp instead of blurred; costs a little of the graphics
# card's time. (The window sets it.)
#aniso=16

# Smooths the edges of the picture on its way to the headset (an FXAA pass): 0 off, 0.35 light,
# 0.65 normal (the default), 1 strong. Rush of Blood draws without multisampling, so this is
# what smooths its edges; a bigger eye_width (supersampling) does it best. (The window sets it.)
#fxaa=0.65

# What the window of the game on the PC shows, which is also what a screen recording of it sees
# (the headset is not affected): left or right, that eye's picture cropped to fill the window like
# a normal game (left is the default); merged, both eyes joined into one flat picture; both, the
# whole picture with both lens circles. pc_zoom, in percent: 100 fits the window with no black
# corners, less is closer in, more shows more of the picture. (The window sets them.)
#pc_view=left
#pc_zoom=100

# The size of the picture handed to the headset's runtime, in percent of the size the game draws
# it at: 100, 75 or 50. It is shrunk to that here, with a smoother filter than Virtual Desktop's.
# Smaller is lighter for the PC and the stream, and a little softer; a large eye_width (2048 and
# more) looks smoother at 50, which is what it is when this is not set (below 2048: 100). (The window sets it, with the Low, Medium and High buttons.)
#layer_scale=100

# Which OpenXR runtime the game uses (also in the start-up window): pc, the one the PC has set
# up (when Virtual Desktop's streamer was the last to set itself, that one); virtualdesktop,
# Virtual Desktop's; steamvr, SteamVR's (SteamVR is started if it is not running, and the
# headset has to be connected to it; not tested with this game yet); or the full path of a
# runtime's .json file.
#runtime=pc

# The most samples a pixel gets (1, 2, 4, 8). Left out: as the console draws it (4). With 1 the
# emulator smooths edges itself instead, unless antialias=0.
#msaa=4
#antialias=1

# 0: your hands are not used to place the controller in the game; it then stays at a fixed
# spot in front of you, turned by its own motion sensors. (The hands are what the headset
# tracks: hand tracking has to be on in the headset, and Virtual Desktop has to pass it on to
# the PC - its setting for forwarding tracking data. Where it does not, the places it gives
# the headset's controllers are used instead, which it makes up from the hands while they lie unused.)
# The DualSense must be connected to this PC itself (USB cable or Bluetooth): paired with the
# headset, it reaches the PC through Virtual Desktop without motion sensors and touchpad.
#hands=1

# How far beyond the next picture the head position given to the game is predicted, in
# milliseconds (0 to 80). More if black edges show when you turn your head, less if the world
# wobbles.
#predict_ms=20

# 0: the right stick no longer doubles as a finger on the touchpad.
#stick_touchpad=1

# 0: the game's surround sound is folded down to left and right instead of being rendered for
# the headset's speakers.
#surround=1

# 0: the game is never made to wait. (It waits, black, while the headset is off the head or
# shows something else, Virtual Desktop's view of the desktop for one, and goes on after.) For
# when it waits although the headset is on and shows the game.
#pause=1

# Not used by this game yet.
#real_time=1

# Not used by this game yet.
#pace=2

# Seconds to wait for a headset before the game starts on the monitor without one. A headset
# connected later is picked up anyway, wherever the game is by then. 0: do not wait.
#wait=60

# 0: do not look for a headset at all, play on the monitor.
#headset=1

# Where the game is, if it is not in the games folder next to this one: its eboot.bin, its
# folder, or its .pkg package (which is then unpacked into the games folder). The launcher
# writes this line itself when it has to ask where the game is.
#game=C:\path\to\CUSA03683\eboot.bin

# Which controllers play the game: dualsense (the default), or quest (the Quest controllers play
# as the game's two PlayStation Move wands). Quest is for the Motion edition only: it needs that
# edition's emulator, and the window greys the choice out without it. (The window sets it.)
#controllers=dualsense

# The Motion edition's Quest controllers, set by three sliders in the window: move_pitch, how
# far the in-game guns are raised above the way the controller points, in degrees (0 to 90);
# move_reach, how far the in-game hands move when yours do, in percent of your own movement (5 to
# 130); and move_turn, how far the guns turn when you turn your wrist, in percent (5 to 130).
# Ignored without the Motion edition.
#move_pitch=35
#move_reach=35
#move_turn=42

# Extra environment variables for the emulator, as many lines as needed.
#env=NAME=value
eye_width=1152
fps=60
fov=120
